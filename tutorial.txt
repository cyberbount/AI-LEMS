9. API ENDPOINTS CHINH
================================================================================

# --- Auth ---
POST /api/auth/login          Dang nhap (form-data: username, password)
POST /api/auth/register       Dang ky tai khoan moi (role=user only)
GET  /api/auth/me             Thong tin nguoi dang nhap

# --- Users ---
GET  /api/users               Danh sach nguoi dung (admin/manager)

# --- Devices ---
GET  /api/devices             Danh sach thiet bi
GET  /api/devices?status=     Loc theo trang thai (available/borrowed/maintenance)
POST /api/devices             Them thiet bi (admin/manager)
PATCH /api/devices/{id}/status  Cap nhat trang thai (admin/manager/technician)

# --- Borrow Requests ---
GET  /api/requests            Danh sach yeu cau muon
POST /api/requests            Tao yeu cau muon (device_id, purpose)
PATCH /api/requests/{id}/status?status=  Cap nhat trang thai
  # Trang thai: pending -> approved -> borrowed -> returned
  # Hoac: pending -> rejected
PATCH /api/requests/{id}/approve    Phe duyet (admin/manager)
PATCH /api/requests/{id}/borrow     Nhan thiet bi
PATCH /api/requests/{id}/return     Tra thiet bi

# --- Maintenance ---
GET  /api/maintenance              Danh sach bao tri
POST /api/maintenance              Tao phieu bao tri (admin/manager/technician)
PATCH /api/maintenance/{id}/complete   Hoan thanh bao tri
GET  /api/maintenance/schedules    Lich bao tri
POST /api/maintenance/schedules    Tao lich bao tri

# --- Stats ---
GET  /api/stats               Thong ke (users, devices, requests, maintenance, usage)

# --- Catalog ---
GET  /api/groups              Nhom thiet bi
POST /api/groups              Them nhom
GET  /api(locations           Vi tri
POST /api/locations           Them vi tri

# --- AI ---
POST /api/ai/chat             Hoi AI (message, history, mode)
  # mode: "chat" | "rag" | "summary" | "inspection_alert"

POST /api/chat                Hoi AI (khong can auth)

# --- Health ---
GET  /health                  Trang thai he thong
